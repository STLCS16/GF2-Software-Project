DEVICES:
C = CLOCK(1);
D1 = DTYPE;
D2 = DTYPE;
D3 = DTYPE;
SW1 = SWITCH(0);
SW2 = SWITCH(0);
SW3 = SWITCH(0);
SW4 = SWITCH(0);
SW5 = SWITCH(0);
SW6 = SWITCH(0);

CONNECTIONS:
C -> D1.CLK;
SW1 -> D1.SET;
SW4 -> D1.CLEAR;
D1.QBAR -> D1.DATA;
D1.Q -> D2.CLK;
SW2 -> D2.SET;
SW5 -> D2.CLEAR;
D2.QBAR -> D2.DATA;
D2.Q -> D3.CLK;
SW3 -> D3.SET;
SW6 -> D3.CLEAR;
D3.QBAR -> D3.DATA;

SIGNALS:
C;
D1.Q;
D2.Q;
D3.Q;

END
